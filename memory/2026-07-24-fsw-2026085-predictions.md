```python
actual_set = set(actual)

print(f"2026085 实际红球:{actual}\n")

for name, picks in pred["predictions"].items():
    for n_pick in [6, 9, 12, 18]:
        key = f"top{n_pick}"
        if key in picks:
            hit = sorted(actual_set & set(picks[key]))
            print(f"{name} Top {n_pick}: 命中 {len(hit)}/{n_pick} = {hit}")
    print()
```
