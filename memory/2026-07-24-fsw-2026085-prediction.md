## 🔄 回测脚本(等 2026085 开奖后用)

```python
import json
import pandas as pd

with open("data/backtest/fsw_2026085_prediction.json") as f:
    pred = json.load(f)

df = pd.read_csv("data/双色球_lottery_data.csv")
red_cols = [f"红球{i}" for i in range(1, 7)]
actual = sorted([int(df[df["issue"]==2026085].iloc[0][c]) for c in red_cols])
actual_set = set(actual)

print(f"2026085 实际:{actual}\n")
for n_pick, key in [(6, "predicted_6_贪心50"), (9, "predicted_9_贪心50"),
                     (12, "predicted_12_贪心50"), (6, "predicted_6_自适应30")]:
    picks = pred["next_period_2026085"][key]
    hits = sorted(actual_set & set(picks))
    label = key.replace("_predicted_", " ").replace("贪心50", "[50 期贪心]").replace("自适应30", "[30 期自适应]")
    print(f"{label} Top {n_pick}: 命中 {len(hits)}/{n_pick} = {hits}")
```
